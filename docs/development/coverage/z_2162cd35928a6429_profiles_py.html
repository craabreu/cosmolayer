22@dataclass(frozen=True) 

23class SigmaProfileTable: 

24 """Area-weighted sigma profiles, at atom or molecule level. 

25 

26 Build with ``from_segments`` (bin segment-level data) or ``aggregate`` 

27 (atom-level table to molecule-level). Direct construction wraps 

28 already-computed arrays. 

29 

30 Parameters 

31 ---------- 

32 areas : np.ndarray 

33 Per-row area, shape ``(n,)``. 

34 charges : np.ndarray 

35 Per-row net (or smoothed equivalent) charge, shape ``(n,)``. 

36 profiles : np.ndarray 

37 Per-row area-fraction sigma profile, shape ``(n, len(grid))``. 

38 grid : SigmaGrid 

39 Grid these profiles are binned on. 

40 centered : bool 

41 Whether each row was centered on its mean charge density before 

42 binning (zero first moment). A property of the data, not the grid. 

43 atom_offsets : np.ndarray | None, optional 

44 Global index of each molecule's first atom. ``None`` (default) 

45 means molecule-level; when set, the table is atom-level and 

46 ``aggregate`` can reassemble it. 

47 """ 

48 

49 areas: NDArray[np.float32] 

50 charges: NDArray[np.float32] 

51 profiles: NDArray[np.float32] 

52 grid: SigmaGrid 

53 centered: bool 

54 atom_offsets: NDArray[np.int64] | None = None

80 def from_segments( # noqa: PLR0913 

81 cls, 

82 sigmas: NDArray[np.float64],

84 segment_offsets: NDArray[np.int64], 

85 *,

87 num_rows: int | None = None, 

88 grid: SigmaGrid = DEFAULT_SIGMA_GRID, 

89 centered: bool = False, 

90 num_threads: int | None = None, 

91 ) -> "SigmaProfileTable": 

92 """Bin segment-level data into per-atom or per-molecule sigma 

93 profiles. 

94 

95 ``sigmas`` is each segment's charge density: pass raw 

96 ``charges / areas``, or an averaged density from 

97 ``average_sigmas_by_molecule``. 

98 

99 Parameters 

100 ---------- 

101 sigmas : np.ndarray 

102 Segment charge density, in e/Å² (raw or averaged). 

103 areas : np.ndarray 

104 Segment areas. 

105 segment_offsets : np.ndarray 

106 Start index of each molecule's segments. Must describe 

107 *exactly* the molecules present in the segment-level arrays; 

108 slice offsets and arrays together. 

109 atom_indices : np.ndarray | None, optional 

110 Global atom index of each segment. ``None`` (default) builds 

111 one profile per molecule. When given, builds one profile per 

112 atom; the result's ``atom_offsets`` is 

113 ``atom_indices[segment_offsets]``. 

114 num_rows : int | None, optional 

115 Number of profile rows. ``None`` (default) uses 

116 ``len(segment_offsets)`` (molecule level) or 

117 ``int(atom_indices.max()) + 1`` (atom level). Pass explicitly 

118 when subsetting. 

119 grid : SigmaGrid, optional 

120 Grid to bin onto, by default ``DEFAULT_SIGMA_GRID``. 

121 centered : bool, optional 

122 If True, center each profile on its mean charge density 

123 before binning. 

124 num_threads : int | None, optional 

125 Thread count. ``None`` (default) uses every CPU core. 

126 

127 Returns 

128 ------- 

129 SigmaProfileTable 

130 Atom-level if ``atom_indices`` is given, else molecule-level. 

131 """ 

132 sigmas = np.asarray(sigmas) 

133 areas = np.asarray(areas) 

134 segment_offsets = np.asarray(segment_offsets) 

135 

136 if atom_indices is None: 

137 row_indices = row_indices_from_offsets(segment_offsets, len(sigmas)) 

138 num_rows = len(segment_offsets) if num_rows is None else num_rows 

139 atom_offsets = None 

140 else: 

141 row_indices = np.asarray(atom_indices) 

142 num_rows = int(row_indices.max()) + 1 if num_rows is None else num_rows 

143 atom_offsets = row_indices[segment_offsets].astype(np.int64) 

144 

145 areas_out = np.zeros(num_rows, dtype=np.float32) 

146 charges_out = np.zeros(num_rows, dtype=np.float32) 

147 profiles_out = np.zeros((num_rows, len(grid)), dtype=np.float32) 

148 assert int(row_indices.max(initial=-1)) < num_rows, ( 

149 "atom_indices/segment_offsets reference a row index >= num_rows; " 

150 "segment_offsets must describe exactly the molecules present in " 

151 "the segment-level arrays" 

152 ) 

153 

154 num_segs = len(sigmas) 

155 num_mols = len(segment_offsets) 

156 accumulator = AtomProfileAccumulator(areas_out, charges_out, profiles_out) 

157 

158 def process_range(start_mol: int, stop_mol: int) -> None: 

159 start_seg = segment_offsets[start_mol] 

160 stop_seg = segment_offsets[stop_mol] if stop_mol < num_mols else num_segs 

161 accumulate_atom_profiles( 

162 accumulator, 

163 sigmas[start_seg:stop_seg], 

164 areas[start_seg:stop_seg], 

165 row_indices[start_seg:stop_seg], 

166 grid, 

167 centered, 

168 ) 

169 

170 run_in_threads(process_range, num_mols, num_threads=num_threads) 

171 

172 return cls( 

173 areas_out, 

174 charges_out, 

175 profiles_out, 

176 grid, 

177 centered, 

178 atom_offsets=atom_offsets, 

179 )

181 def aggregate( 

182 self, 

183 *, 

184 grid: SigmaGrid | None = None, 

185 normalize: bool = False, 

186 num_threads: int | None = None, 

187 ) -> "SigmaProfileTable": 

188 """Reassemble per-molecule profiles from these per-atom ones. 

189 

190 Area-weighted sum of atom profiles onto a shared molecule axis. 

191 If this table is centered, each atom is translated back by its 

192 mean charge density first. The result is always uncentered. 

193 

194 Parameters 

195 ---------- 

196 grid : SigmaGrid | None, optional 

197 Grid for the molecule profiles. ``None`` (default) uses 

198 ``self.grid``. A different grid must share this table's 

199 ``bin_width``. 

200 normalize : bool, optional 

201 If True, divide each molecule's profile by its total area so 

202 it sums to 1. 

203 num_threads : int | None, optional 

204 Thread count. ``None`` (default) uses every CPU core. 

205 

206 Returns 

207 ------- 

208 SigmaProfileTable 

209 Molecule-level (``atom_offsets`` is None, ``centered`` False). 

210 

211 Raises 

212 ------ 

213 ValueError 

214 If this table is already molecule-level, or if ``grid`` does 

215 not share this table's ``bin_width``. 

216 """ 

217 if self.atom_offsets is None: 

218 raise ValueError( 

219 "This SigmaProfileTable has no atom_offsets -- it is " 

220 "already at molecule level, so there is nothing to " 

221 "aggregate." 

222 ) 

223 output_grid = self.grid if grid is None else grid 

224 if not math.isclose(output_grid.bin_width, self.grid.bin_width, rel_tol=1e-9): 

225 raise ValueError( 

226 f"Output grid bin width {output_grid.bin_width!r} does not " 

227 f"match this table's {self.grid.bin_width!r}. Aggregation " 

228 "aligns columns by point-count difference alone, which is " 

229 "only valid between symmetric grids of equal bin width." 

230 ) 

231 

232 num_atoms = len(self.areas) 

233 num_mols = len(self.atom_offsets) 

234 molecule_num_points = len(output_grid) 

235 

236 molecule_indices = row_indices_from_offsets(self.atom_offsets, num_atoms) 

237 

238 translations = np.zeros(num_atoms, dtype=np.float64) 

239 if self.centered: 

240 has_area = self.areas > 0 

241 translations[has_area] = self.charges[has_area] / self.areas[has_area] 

242 

243 molecule_profiles = np.zeros((num_mols, molecule_num_points), dtype=np.float64) 

244 atom_offsets = self.atom_offsets 

245 

246 def process_range(start_mol: int, stop_mol: int) -> None: 

247 start_atom = atom_offsets[start_mol] 

248 stop_atom = atom_offsets[stop_mol] if stop_mol < num_mols else num_atoms 

249 batch = AtomTranslationBatch( 

250 self.areas[start_atom:stop_atom], 

251 translations[start_atom:stop_atom], 

252 self.profiles[start_atom:stop_atom], 

253 self.grid, 

254 ) 

255 accumulate_translated_profiles( 

256 molecule_profiles, 

257 molecule_indices[start_atom:stop_atom], 

258 batch, 

259 output_grid, 

260 ) 

261 

262 run_in_threads(process_range, num_mols, num_threads=num_threads) 

263 

264 if normalize: 

265 molecule_profiles = molecule_profiles / molecule_profiles.sum( 

266 axis=1, keepdims=True 

267 ) 

268 

269 molecule_areas = compute_per_molecule_properties(self.areas, atom_offsets) 

270 molecule_charges = compute_per_molecule_properties(self.charges, atom_offsets) 

271 

272 return replace( 

273 self, 

274 areas=molecule_areas, 

275 charges=molecule_charges, 

276 profiles=molecule_profiles.astype(np.float32), 

277 grid=output_grid, 

278 centered=False, 

279 atom_offsets=None, 

280 )
